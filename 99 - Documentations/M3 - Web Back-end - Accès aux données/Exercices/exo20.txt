// function présentation
function etatCivil($personnes)
{
    $sommeNotes = 0;
    $moyenneNotes = 0;
    $tirets = 1;

    foreach ($personnes as $personne) {
        foreach ($personne as $clefPersonne => $vPersonne) {
            if ($clefPersonne != 'sexe' && $clefPersonne != 'notes') {
                echo $clefPersonne . ' : ' . $vPersonne . '<br>';
            } elseif ($clefPersonne == 'sexe') {
                echo $clefPersonne . ' : ' . ($vPersonne ? 'homme' : 'femme') . '<br>';
            } else {
                foreach ($vPersonne as $clefVPersonne => $notes) {
                    $sommeNotes += $notes;
                }
                $moyenneNotes = round($sommeNotes / $clefVPersonne, 2);
                echo $clefPersonne . ' : ' . $moyenneNotes . '<br>';

                $sommeNotes = 0;
            }
        }

        echo $tirets < count($personnes) ? "==============<br>" : '';
        $tirets++;
    }
}
